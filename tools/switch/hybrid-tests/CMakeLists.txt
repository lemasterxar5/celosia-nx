cmake_minimum_required(VERSION 3.21)
project(CelosiaHybridHostTests LANGUAGES CXX)
enable_testing()
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(platform "${root}/PS2Recomp/ps2xRuntime/platform/switch")
find_package(Threads REQUIRED)
add_executable(hybrid_sound_test "${platform}/tests/hybrid_sound_test.cpp" "${platform}/hybrid/raylib_sounds.cpp")
add_executable(hybrid_window_test "${platform}/tests/hybrid_window_test.cpp" "${platform}/hybrid/raylib_window.cpp")
target_include_directories(hybrid_window_test PRIVATE "${platform}/tests/host_switch" "${root}/PS2Recomp/ps2xRuntime/include")
add_executable(native_texture_asset_test "${platform}/tests/native_texture_asset_test.cpp" "${platform}/native_texture_asset.cpp")
add_executable(hybrid_owned_backend_test "${platform}/tests/hybrid_owned_backend_test.cpp" "${platform}/hybrid/owned_backend.cpp"
    "${platform}/native_executor.cpp" "${platform}/native_threads.cpp" "${platform}/native_semaphore.cpp")
target_include_directories(hybrid_owned_backend_test PRIVATE "${root}/PS2Recomp/ps2xRuntime/src/lib/native" "${root}/PS2Recomp/ps2xRuntime/include")
add_executable(hybrid_game_thread_test "${platform}/tests/hybrid_game_thread_test.cpp")
add_executable(hybrid_progress_monitor_test "${platform}/tests/hybrid_progress_monitor_test.cpp")
set(deko "${root}/PS2Recomp/ps2xRuntime/src/lib/native/deko3d")
add_executable(deko_arena_test "${deko}/tests/arena_test.cpp")
add_executable(deko_page_pool_test "${deko}/tests/page_pool_test.cpp")
add_executable(deko_gpu_timing_test "${deko}/tests/gpu_timing_test.cpp")
add_executable(deko_frame_rate_test "${deko}/tests/frame_rate_test.cpp")
add_executable(deko_policy_test "${deko}/tests/policy_test.cpp")
add_executable(deko_pipeline_test "${deko}/tests/pipeline_test.cpp")
add_executable(deko_late_mesh_test "${deko}/tests/late_mesh_test.cpp")
add_executable(deko_opaque_policy_test "${deko}/tests/opaque_policy_test.cpp")
add_executable(deko_descriptor_slots_test "${deko}/tests/descriptor_slots_test.cpp")
add_executable(deko_prepass_policy_test "${deko}/tests/prepass_policy_test.cpp")
add_executable(deko_gpu_segments_test "${deko}/tests/gpu_segments_test.cpp")
target_include_directories(deko_prepass_policy_test PRIVATE "${root}/PS2Recomp/ps2xRuntime/include")
target_include_directories(deko_opaque_policy_test PRIVATE "${root}/PS2Recomp/ps2xRuntime/include")
target_include_directories(deko_policy_test PRIVATE "${root}/PS2Recomp/ps2xRuntime/include")
target_include_directories(deko_pipeline_test PRIVATE "${root}/PS2Recomp/ps2xRuntime/include")
foreach(target IN ITEMS hybrid_sound_test hybrid_window_test native_texture_asset_test hybrid_owned_backend_test hybrid_game_thread_test hybrid_progress_monitor_test deko_arena_test deko_page_pool_test deko_gpu_timing_test deko_frame_rate_test deko_policy_test deko_pipeline_test deko_late_mesh_test deko_opaque_policy_test deko_descriptor_slots_test deko_prepass_policy_test deko_gpu_segments_test)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${platform}/hybrid" "${platform}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    add_test(NAME ${target} COMMAND ${target})
endforeach()
